character_event = { #Selection between ways of Caesar Reborn proclamation.
    id = CCRE.2
	hide_window = yes
	capable_only = yes
	is_triggered_only = yes
	
	immediate = {
	    if = { #Pontifex Maximus proclaim that Emperor is a Caesar Reborn
			limit = {
				ROOT = { NOT = { controls_religion = yes } }
				d_cult_of_mars = { has_owner = yes }
			}
			d_cult_of_mars = {
				owner = { 
				    event_target:caesar_reborn_candidate = { letter_event = { id = CCRE.4 days = 5 } }
				}
			}
		}
		if = { #Pontifex Maximus IS the Emperor, so he proclaim that he is a Caesar Reborn himself
			limit = {
				ROOT = { controls_religion = yes }
				d_cult_of_mars = { has_owner = yes }
			}
			ROOT = {
				character_event = { id = CCRE.3 days = 5 }
			}
		}
		if = { #Pontifex Maximus does not exist, so Emperor proclaim that he is a Caesar Reborn himself
			limit = {
				d_cult_of_mars = { NOT = { has_owner = yes } }
			}
			ROOT = {
				character_event = { id = CCRE.3 days = 5 }
			}
		}
	}
	
}
