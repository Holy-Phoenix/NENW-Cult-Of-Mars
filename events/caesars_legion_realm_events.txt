narrative_event = {
    id = CLRE.3
    title = EVTTITLE_CLRE.3
    desc = {
	    trigger = {
            is_roman_gods_cultist_trigger = yes
        }
	    text = EVTDESC_CLRE_A.3
	}
	desc = {
	    trigger = {
            religion_group = christian
        }
	    text = EVTDESC_CLRE_B.3
	}
	desc = {
	    trigger = {
            is_russian_communist_religion_trigger = yes
        }
	    text = EVTDESC_CLRE_C.3
	}
	desc = {
	    trigger = {
            religion_openly_novelist_or_reformed_trigger = yes
        }
	    text = EVTDESC_CLRE_D.3
	}
    picture = GFX_evt_roman_triumph
    border = GFX_event_narrative_frame_diplomacy
    is_triggered_only = yes

    option = {      
        name = CLREO.1
		trigger = {
			has_landed_title = e_nova_roma
		}
		if = {
		    limit = {
			    religion = cult_of_mars
			}
			add_trait = caesar_reborn
			set_character_flag = caesar_reborn_trait_holder
		    set_global_flag = caesar_reborn_existing
		}
		give_nickname = nick_the_glorious
		if = {
			limit = {
				is_female = yes
			}
			create_bloodline = {
				type = martian_emperor_female
			}
		}
		if = {
			limit = {
				is_female = no
			}
			create_bloodline = {
				type = martian_emperor
			}
		}
		sound_effect = bloodline_added
		set_government_type = roman_imperial_martian_government 
		narrative_event = { id = CLRE.9 }
    }
	
	option = {      
        name = CLREO.2
		trigger = {
			NOT = { has_landed_title = e_nova_roma }
		}
    }
}

# What would we do with Caesar's Legion after we proclaim Roman Empire?
narrative_event = {
	id = CLRE.9
	title = CLRE.9.title
	desc = CLRE.9.desc
	picture = GFX_evt_legion2
	border = GFX_event_narrative_frame_war

	is_triggered_only = yes
	
	trigger = {
		OR = {
			primary_title = { title = e_nova_roma }
		}
		OR = {
		    k_caesar_legion = { has_holder = ROOT }
		}
	}

	option = { #Reform it to Imperial Army
		name = EVTOPTACLRE9
		set_global_flag = imperial_army_founded
		set_global_flag = legion_disbanded
		if = {
			limit = {
				primary_title = { title = e_nova_roma }
			}		    
			activate_title = { title = d_imperial_army status = yes }
			hidden_tooltip = {
					create_character = {
					random_traits = yes
					religion = cult_of_mars
					culture = roman
					dynasty = NONE
					female = no
					age = 34
					trait = brilliant_strategist
				}
				new_character = {
					d_imperial_army = {
						grant_title = PREV
					}
					set_defacto_liege = ROOT
				}
			}
		}
         hidden_tooltip = {
			activate_title = { title = k_caesar_legion status = no }
		}
		k_caesar_legion = {
			show_scope_change = no
			destroy_landed_title = THIS
		}
        narrative_event = { id = CLRE.11 days = 5 }		
	}

	option = { #Disband it
		name = EVTOPTBCLRE9
		set_global_flag = legion_disbanded
		hidden_tooltip = {
			activate_title = { title = k_caesar_legion status = no }
		}
		k_caesar_legion = {
			show_scope_change = no
			destroy_landed_title = THIS
		}
	}
}
