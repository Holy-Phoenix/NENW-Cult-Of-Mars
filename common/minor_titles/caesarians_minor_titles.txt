#Roman Republic titles
title_senator_caesarian = {
	dignity = 3
	realm_in_name = yes
	show_as_title = yes
	grant_limit = 12
	is_voter = yes
	is_high_prio = yes
	
	monthly_salary = 0.075
	monthly_prestige = 0.5
	

	
	allowed_to_hold = {
		OR = {
			male_can_hold_minor_title_trigger = yes
			female_can_hold_minor_title_trigger = yes
			has_law = true_cognatic_succession
		}
		is_adult = yes
		prisoner = no
		is_incapable = no
		has_regent = no
		NOT = { trait = monk }
		NOT = { trait = nun }
	}
	
	allowed_to_grant = {
		OR = {
		    has_landed_title = e_first_rome
			has_landed_title = e_second_rome
			has_landed_title = e_third_rome
			has_landed_title = e_first_rome_republic
			has_landed_title = e_second_rome_republic
			has_landed_title = e_third_rome_republic
        }
		primary_title = { 
			OR = {
				has_law = parliament_laws_roman_caesarian_1 
				has_law = parliament_laws_roman_caesarian_2
				has_law = parliament_laws_roman_caesarian_3
			}
			NOT = { 
				has_law = parliament_laws_roman_caesarian_0
			}
		}
	}
	
	gain_effect = {
		character_event = { id = DRUK.7 }
	}
	lose_effect = {
		character_event = { id = DRUK.8 }
		if = {
			limit = {
				FROM = {
					NOT = {
						has_character_flag = swapping_councillors
					}
				}
			}
			opinion = { who = FROM modifier = opinion_fired_from_council }
			opinion = { who = FROM modifier = opinion_recently_fired }
		}
	}

	message = yes
}
